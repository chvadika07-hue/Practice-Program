class Student:
    def __init__(self, name, roll_no):
        self.name = name
        self.roll_no = roll_no
        self.courses = []

class Course:
    def __init__(self, course_name):
        self.course_name = course_name
        self.students = []

class College:
    def __init__(self):
        self.students = []
        self.courses = []

    def add_student(self, student):
        self.students.append(student)

    def add_course(self, course):
        self.courses.append(course)

    def enroll_student(self, student, course):
        if course not in student.courses:
            student.courses.append(course)
            course.students.append(student)

    def remove_student(self, student, course):
        if course in student.courses:
            student.courses.remove(course)
            course.students.remove(student)


# Example
college = College()

student1 = Student("Sofia", 101)
student2 = Student("Anna", 102)

course1 = Course("Python")
course2 = Course("Data Structures")

college.add_student(student1)
college.add_student(student2)

college.add_course(course1)
college.add_course(course2)

college.enroll_student(student1, course1)
college.enroll_student(student2, course2)
